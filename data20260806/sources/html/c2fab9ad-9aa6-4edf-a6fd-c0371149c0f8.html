5.3.解决报错“语句句柄个数超过上限或系统内存不足”
修改达梦数据库参数 dm.ini
参数
建议值
说明
MAX_SESSION_STATEMENT
20000
每个会话打开的最大句柄数
由于程序很多地方存在不关闭 statement 的问题，导致达梦数据库报错“语句句柄个数超过上限或系统内存不足”。为避免此问题，建议从程序入手根本解决。
可以用下列语句查询会话打开的句柄：
select session#, sql_str from v$open_stmt where sql_str is not null order by sql_str;
select session#, count(*) cnt from v$open_stmt group by session# order by cnt desc;

begin
SELECT
TOP 1 N_CPU,
TOTAL_PHY_SIZE/1024/1024
INTO
v_cpus,
v_mem_mb
FROM
V$SYSTEMINFO;

print v_mem_mb;

print v_mem_mb;

TASK_THREADS :=4;

SP_SET_PARA_VALUE(2, 'TASK_THREADS', TASK_THREADS);
BUFFER := round(cast(v_mem_mb * 0.5 as int), -3);
MAX_BUFFER := BUFFER;
RECYCLE :=LEAST(cast(v_mem_mb * 0.125 as int), 50000);
IF v_mem_mb < 70000 THEN
--设置根据内存情况BUFFER_POOLS参数
with
t as
(
select rownum rn from dual connect by level <= 100
)
,
t1 as
(
select * from t where rn > 1 minus
select
ta.rn * tb.rn
from
t ta,
t tb
where
ta.rn <= tb.rn
and ta.rn > 1
and tb.rn > 1
)
select top 1 rn into BUFFER_POOLS from t1 where rn > v_mem_mb/800 order by 1;
PRINT 'BUFFER_POOLS ' || BUFFER_POOLS;
--设置根据内存情况RECYCLE_POOLS参数
with
t as
(
select rownum rn from dual connect by level <= 100
)
,
t1 as
(
select * from t where rn > 1 minus
select
ta.rn * tb.rn
from
t ta,
t tb
where
ta.rn <= tb.rn
and ta.rn > 1
and tb.rn > 1
)
select
top 1 rn
into
RECYCLE_POOLS
from
t1
where
rn > v_mem_mb/800/3
order by
1;
PRINT 'RECYCLE_POOLS ' || RECYCLE_POOLS;
ELSE
BUFFER_POOLS := 101;
RECYCLE_POOLS := 41;
END IF;

MEMORY_POOL := 2048;
CACHE_POOL_SIZE := 1024;

SORT_BLK_SIZE =1;

HAGR_BUF_SIZE :=500;

6.表行数统计
declare
stmt varchar2(4000);
tablename varchar2(1000);
nums int;
begin
for p in (select table_name from dba_tables where owner='EZOFFICE' order by table_name) loop
stmt := 'select '''||p.table_name||''',count(*) from ezoffice.'||p.table_name;
execute immediate stmt into tablename,nums;
dbms_output.put_line(tablename||','||nums);
end loop;
end;
